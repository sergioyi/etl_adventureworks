cursor_postgres.execute("""
CREATE TABLE IF NOT EXISTS staging.FatoVendas (
  IdPedido INT,
  IdProduto INT,
  IdCliente INT,
  IdVendedor INT,
  IdRegiao INT,
  IdData INT,
  DataEnvio DATE,
  Quantidade INT,
  PrecoUnitario DECIMAL(10,2),
  Receita DECIMAL(12,2),
  Custo DECIMAL(12,2),
  Lucro DECIMAL(12,2),
  PRIMARY KEY (IdPedido, IdProduto),
  FOREIGN KEY (IdProduto) REFERENCES DimProduto(IdProduto),
  FOREIGN KEY (IdCliente) REFERENCES DimCliente(IdCliente),
  FOREIGN KEY (IdVendedor) REFERENCES DimVendedor(IdVendedor),
  FOREIGN KEY (IdRegiao) REFERENCES DimRegiao(IdRegiao),
  FOREIGN KEY (IdData) REFERENCES DimTempo(IdData) );
  """)
cursor_postgres.execute("""
CREATE TABLE IF NOT EXISTS staging.FatoEstoque (
  IdProduto INT,
  IdData INT,
  QuantidadeEstoque INT,
  PRIMARY KEY (IdProduto, IdData),
  FOREIGN KEY (IdProduto) REFERENCES DimProduto(IdProduto),
  FOREIGN KEY (IdData) REFERENCES DimTempo(IdData) );
  """)

cursor_postgres.commit()
